



finding sqlite3 rows

sqlite uses a varint format where if bit 8 (&0x80) is set, another byte follows.

rows consist of header data followed by content. every row has a type header! it contains lengths.

- varint headerlen
for each element:
	- varint serial type
		0 = NULL
		<4 = integer of that many bytes
		5,6 = integer of 6, 8 bytes
		7 = float of 8 bytes
		8 = elided constant 0
		9 = elided constant 1
		[10 = NULL with virtual table]
		>=12 and even, blob of size (n-12)/2
		>=13 and odd, text of size (n-13)/2
the data then procedes packed after the header.

so, record:
varint reclen
varint id
varint hdrlen
	repeated:
		serial type as above
repeated:
	data as serial type
next record

so if i find myself in a record, i roughly need to walk backward hypothesizing the content.
if i know the table structure, i can interpret the data i see as that table structure, and compare it with the content.
if i make a serial type parser that can return failure, then it's not hard.

int parse_sqlite_null(unsigned char ** hdrdata, unsigned char ** data, void ** null = 0)
{
	if (*(hdrdata ++) != 0) return 0;
	if (null) *null = 0;
	return 1;
}

int parse_sqlite_int(unsigned char ** hdrdata, unsigned char ** data, uint64_t * integer)
{
	uint32_t type;
	uint32_t int_len;
	if (!parse_varint(hdrdata, &type)) return 0;
	if (type == 0 || type == 8 || type >= 10) return 0;
	if (type < 5) {
		int_len = type;
	} else if (type < 7) {
		int_len = (type - 2) * 2;
	} else {
		*integer = type & 1;
		return 1;
	}
	unsigned char * data_tail = *data + int_len;
	*integer = 0;
	assert(*(uint16_t*){0x01,0x02} == 0x201); // little endian
	memcpy(integer, *data, int_len);
	*data += int_len;
}

int parse_sqlite_float(unsigned char ** hdrdata, unsigned char ** data, double * floating)
{
	if ((*hdrdata) ++ != 7) return 0;
	*floating = *(double*)*data;
	*data += 8;
	return 1;
}

int parse_sqlite_text(unsigned char ** hdrdata, unsigned char ** data, char ** text, uint32_t * text_len)
{
	uint32_t type;
	if (!parse_varint(hdrdata, &type)) return 0;
	if (type < 13 || !(type & 1)) return 0;
	*text_len = (type - 13) / 2;
	*text = data;
	*data += text_len;
	return 1;
}

int parse_sqlite_blob(unsigned char ** hdrdata, unsigned char ** data, unsigned char ** blob, uint32_t * blob_len)
{
	uint32_t type;
	if (!parse_varint(hdrdata, &type)) return 0;
	if (type < 12 || (type & 1)) return 0;
	*blob_len = (type - 12) / 2;
	*blob = data;
	*data += blob_len;
	return 1;
}

parse_sqlite_record(data) {
	unsigned char * recptr = data;
	unsigned char * hdrptr = recptr;
	unsigned char * hdrdata, dataptr;
	int hdroffset = 0;
	uint32_t reclen, id, hdrlen;
	if (!parse_varint(&hdrptr, &reclen)) return 0;
	if (!parse_varint(&hdrptr, &id)) return 0;
	if (!parse_varint(&hdrptr, &hdrlen)) return 0;

	hdrdata = hdrptr;
	dataptr = hdrptr + hdrlen;

	// id
	if (!parse_sqlite_null(&hdrdata, &dataptr)) return 0;

	// path
	char * path; size_t path_len
	if (!parse_sqlite_text(&hdrdata, &dataptr, &path, &path_len)) return 0;

	// hash
	char * hash; size_t hash_len
	if (!parse_sqlite_text(&hdrdata, &dataptr, &hash, &path_len)) return 0;
	if (hash_len != 71) return 0;

	// registrationTime
	uint64_t registrationTime;
	if (!parse_sqlite_int(&hdrdata, &dataptr, &registrationTime)) return 0;

	// deriver
	char * deriver; size_t deriver_len;
	if (parse_sqlite_null(&hdrdata, &dataptr, &deriver)) {
		deriver_len = 0;
	} else if (!parse_sqlite_text(&hdrdata, &dataptr, &deriver, &deriver_len)) {
		return 0;
	}

	// narSize
	uint64_t narSize;
	if (parse_sqlite_null(&hdrdata, &dataptr)) {
		narSize = ~0;
	} else if (!parse_sqlite_int(&hdrdata, &dataptr, &narSize)) {
		return 0;
	}

	return 1;
}
